bool isInRange(pair <ld, ld> p, ld a) {
    return p.f - eps <= p.s ? (p.f - eps <= a && a <= p.s + eps) : (p.f - eps <= a || a <= p.s + eps);
}

pair <ld, ld> mergeRanges(pair <ld, ld> a, pair <ld, ld> b) { // (a.s - a.f) % 2PI <= Pi, (b.s - b.f + 2PI) % 2PI <= PI
    if (isInRange(b, a.f))
        return min(pair <ld, ld>{a.f, a.s}, pair <ld, ld>{a.f, b.s}, [](pair <ld, ld> a, pair <ld, ld> b) {
            return a.s - a.f + 2 * PI * (a.s < a.f) < b.s - b.f + 2 * PI * (b.s < b.f);
        });
    if (isInRange(b, a.s))
        return min(pair <ld, ld>{a.f, a.s}, pair <ld, ld>{b.f, a.s}, [](pair <ld, ld> a, pair <ld, ld> b) {
            return a.s - a.f + 2 * PI * (a.s < a.f) < b.s - b.f + 2 * PI * (b.s < b.f);
        });
    if (isInRange(a, b.f))
        return b;
    return {-1, -1};
}

ld distToSegment(int x0, int y0, int x1, int y1, int x2, int y2) { // From (x0, y0) to (x1, y1)-(x2, y2)
    ld a = sqrtl(dist2(x0, y0, x1, y1)),
       b = sqrtl(dist2(x0, y0, x2, y2)),
       c = sqrtl(dist2(x1, y1, x2, y2));
    ld x = (b * b - a * a - c * c) / c / -2;
    if (0 <= x && x <= c)
        return sqrt(a * a - x * x);
    return 9e18; // Dist to point
}

Код для антивзлома xor-хешей + корнячка через gp_hash_table для операций xor^= на отрезке и поиска последнего вхождения x (быстрее невозможно):
using u64 = unsigned long long;
struct H { size_t operator()(u64 x) const { // splitmix64, против анти-хеш тестов
    x += 0x9e3779b97f4a7c15; x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9;
    x = (x ^ (x >> 27)) * 0x94d049bb133111eb; return x ^ (x >> 31); } };
const int B = 512;
struct Block { vector<u64> c; u64 lz = 0; gp_hash_table<u64, int, H> last; };
vector<Block> bl; int n = 0;

void rebuild(int k) {
    auto& b = bl[k]; b.last.clear();
    for (int j = 0; j < (int)b.c.size(); j++) b.last[b.c[j]] = k * B + j; // по возрастанию => остаётся максимум
}
void push(u64 h) {
    if (n % B == 0) bl.emplace_back();
    auto& b = bl.back(); u64 v = h ^ b.lz;
    b.c.push_back(v); b.last[v] = n++;
}
void suffixXor(int l, u64 x) {               // 0-индексация
    int k = l / B;
    for (int j = l - k * B; j < (int)bl[k].c.size(); j++) bl[k].c[j] ^= x;
    rebuild(k);
    for (int t = k + 1; t < (int)bl.size(); t++) bl[t].lz ^= x;
}
int findLast(u64 x) {
    for (int k = (int)bl.size() - 1; k >= 0; k--) {
        auto it = bl[k].last.find(x ^ bl[k].lz);
        if (it != bl[k].last.end()) return it->second;
    }
    return -1;
}

Геометрия на векторах с ЦЕЛЫМИ координатами: всё проверяется точно, без atan2, sqrt, деления и eps
Вектор a = (ax, ay) - стрелка из (0, 0) в точку (ax, ay), ax и ay - целые (long). Вектор из точки P в точку Q: Q - P = (Qx - Px, Qy - Py), у целых точек он тоже целый.
Сложение, вычитание и умножение целых чисел дают ответ без погрешности. Погрешность появляется только от sqrt, atan2, деления и сравнений с eps, поэтому их не используем.
    1. Квадрат длины: |a|^2 = ax^2 + ay^2 (теорема Пифагора). Сама длина sqrt(ax^2 + ay^2) бывает иррациональной, поэтому длины и расстояния сравниваем по квадратам: |a| < |b| <=> |a|^2 < |b|^2 (обе длины >= 0).
        Пример: |(3, 4)|^2 = 9 + 16 = 25, |(1, 7)|^2 = 1 + 49 = 50 - (1, 7) длиннее.
    2. Скалярное произведение: dot(a, b) = ax * bx + ay * by = |a| * |b| * cos(угла между a и b). dot(a, a) = |a|^2.
        Знак показывает угол между a и b: > 0 - острый, = 0 - прямой (a и b перпендикулярны), < 0 - тупой.
        Почему dot = 0 <=> прямой угол: в треугольнике с вершинами 0, a, b третья сторона - вектор a - b. Раскроем скобки:
            |a - b|^2 = (ax - bx)^2 + (ay - by)^2 = ax^2 + ay^2 + bx^2 + by^2 - 2 * (ax * bx + ay * by) = |a|^2 + |b|^2 - 2 * dot(a, b).
            По теореме Пифагора и обратной к ней угол при 0 прямой <=> |a - b|^2 = |a|^2 + |b|^2 <=> dot(a, b) = 0. (Это теорема косинусов, отсюда и cos в формуле.)
        Пример: (3, 4) и (-4, 3): 3 * (-4) + 4 * 3 = 0 - перпендикулярны. (1, 0) и (1, 1): 1 > 0 - угол 45 градусов, острый. (1, 0) и (-1, 1): -1 < 0 - угол 135 градусов, тупой.
    3. Поворот на 90 градусов против часовой стрелки: rot(x, y) = (-y, x); по часовой: -rot(x, y) = (y, -x); на 180 градусов: (-x, -y).
        Длина не меняется: (-y)^2 + x^2 = x^2 + y^2. Перпендикулярность: dot(a, rot(a)) = ax * (-ay) + ay * ax = 0.
        Пример: rot(3, 1) = (-1, 3): dot((3, 1), (-1, 3)) = -3 + 3 = 0, квадраты длин оба 10.
    4. Векторное (косое) произведение: cross(a, b) = ax * by - ay * bx = |a| * |b| * sin(угла поворота от a к b против часовой стрелки).
        Знак: > 0 - b левее a (от a к b поворот против часовой), < 0 - правее (по часовой), = 0 - a и b на одной прямой через 0 (коллинеарны: сонаправлены или противоположны).
        Почему: cross(a, b) = (-ay) * bx + ax * by = dot(rot(a), b). Значит, cross = 0 <=> b перпендикулярен rot(a) <=> b параллелен a, а знак показывает, в ту ли сторону от a смотрит b, что и rot(a) (влево).
        Порядок важен: cross(b, a) = -cross(a, b).
        |cross(a, b)| - площадь параллелограмма на векторах a и b, половина - площадь треугольника 0, a, b. Площадь треугольника с целыми вершинами кратна 1/2, поэтому храним удвоенную площадь (целую).
        Для точек A, B, C знак cross(B - A, C - A) говорит, где C относительно прямой AB (если идти из A в B): слева, справа или на ней.
        Пример: (2, 0) и (1, 3): 2 * 3 - 0 * 1 = 6 > 0 - (1, 3) левее (2, 0), площадь треугольника 0, (2, 0), (1, 3) равна 3. (2, 4) и (-1, -2): 2 * (-2) - 4 * (-1) = 0 - коллинеарны (противоположны).
    5. Точка P на отрезке AB: cross(B - A, P - A) == 0 (P на прямой AB) и dot(P - A, P - B) <= 0 (из P векторы в A и B смотрят в противоположные стороны, либо P совпадает с концом). Строго внутри отрезка: dot < 0.
        Частный случай: точка 0 строго между точками A и B: cross(A, B) == 0 && dot(A, B) < 0.
        Пример: A = (0, 0), B = (4, 2), P = (2, 1): cross((4, 2), (2, 1)) = 4 - 4 = 0, dot((2, 1), (-2, -1)) = -5 < 0 - внутри. P = (6, 3): cross = 0, dot((6, 3), (2, 1)) = 15 > 0 - на прямой, но за B.
    6. Квадрат по стороне AB: d = rot(B - A), вершины по порядку A, B, B + d, A + d. Квадрат лежит слева от AB, если идти из A в B; для правого берём d = -rot(B - A). Все вершины целые, корни не нужны.
        Пример: A = (0, 0), B = (2, 1): d = (-1, 2), квадрат (0, 0), (2, 1), (1, 3), (-1, 2).
        Квадрат по диагонали AC (вершины A, B, C, D против часовой): B = (A + C + rot(A - C)) / 2, D = (A + C - rot(A - C)) / 2 (поворот A вокруг центра (A + C) / 2). Вершины целые <=> числители чётные.
        Пример: A = (0, 0), C = (2, 2): rot(A - C) = rot(-2, -2) = (2, -2), B = (4, 0) / 2 = (2, 0), D = (0, 4) / 2 = (0, 2).
    7. Целые перпендикулярные векторы. Пусть g = gcd(|ax|, |ay|), тогда (ax / g, ay / g) - самый короткий целый вектор направления a. Любой целый b, перпендикулярный a, равен k * (-ay / g, ax / g) с целым k (так как ax / g и ay / g взаимно просты).
        Следствие: точка на луче b на расстоянии |a| от 0 - это rot(a) или -rot(a) (тот, у которого dot с b > 0). Её можно найти без корней, и она всегда целая.
        Пример: a = (4, 2), b = (-3, 6): dot = -12 + 12 = 0, g = 2, b = 3 * (-1, 2). rot(a) = (-2, 4), dot((-2, 4), b) = 6 + 24 = 30 > 0 - это и есть точка на луче b на расстоянии |a|.
    8. Полярная сортировка (по углу от оси OX против часовой стрелки) без atan2. Делим на две полуплоскости: half(a) = (ay < 0 || ay == 0 && ax < 0) (0 - углы [0, 180), 1 - углы [180, 360)).
        a раньше b <=> half(a) < half(b), либо half(a) == half(b) и cross(a, b) > 0. Внутри полуплоскости все углы меньше 180 градусов, поэтому сравнение через cross корректно.
        Пример: (1, 0) < (1, 1) < (0, 1) < (-1, 1) < (-1, 0) < (0, -1) < (1, -1).
    9. Переполнение: если |координаты| <= C, то разности по модулю <= 2C, а dot и cross <= 2 * (2C)^2 = 8C^2. При C = 1e9 это 8e18 - еле влезает в long long (до 9.2e18). Если координаты больше или результат ещё раз умножается, то __int128.

Полиномы:
Критерий Эйзенштейна:
Полином неприводим (не расскладывается на 2 полинома меньшей степени в целых числах (либо в рациональных (p / q), домноженных на НОК q)), если старший коэффициент (макс. степени) не делится на простое число P, все остальные коэффициенты делятся на P, а также свободный член не делится на P^2. (x^3 + 2 неприводим). Однако, его отсутствие на гарантирует, что полином приводим.
Неприводимые полиномы степени больше 1 (не x - a) не имеют рациональных корней (вещественные могут: x^2 - 2 пересекает OX), следовательно, уравнения огромных степеней можно решать, выделяя делители (q x - p), где p | a_0, q | a_n, пока полином остаётся приводимым.
Приводимый полином (в целых числах, для рац. тоже самое), у которого коэффициенты имеют gcd = 1 (нельзя выделить константу, на которую умножать), если он имеет 2 разложения, отличающиеся не только порядком множителей, имеют приводимый множитель:
6x^2 + 6 = 6 * (x^2 + 1) (приводимый 1 способом, x^2 + 1 неприводим; (x - 1)(x + 1) = x^2 - 1)
x^3 - 6x^2 + 11x - 6 = (x - 1)(x - 2)(x - 3) = (x - 1)(x^2 - 5x + 6) = (x - 2)(x^2 - 4x + 3) (приводим несколькими способами, но только одним из неприводимых многочленов)
Если полином P(x) приводим, то и полином P(x^k) тоже приводим (все степени умножились на k, коэффициенты перешли с ними)
Если критерий Эйзенштейна сработал при P(x + k), значит, что P(x) также неприводим - тестировать его на нескольких k
Если мы знаем, что a ^ b = x и a + b <= s, то кол-во таких неотрицательных целых a, b можно найти ДП по числу:
a + b = a ^ b + (a & b) * 2
x + (a & b) * 2 <= s
a & b <= (s - x) / 2
Находим кол-во таких пар, у которых на битах x не стоит 1 (т. е. только одинаковые биты a и b) и строим  dp[i][equal].
(dp[0][0] + dp[0][1]) * 2 ** popcount(x)

Если нужно пометить вершины мин кол-вом точек в двудольном графе, чтобы каждое ребро было соединено с хотя-бы одной помеченной вершиной: их кол-во равно макс. парсочу. Если у вершины ребра из макс. парсоча есть рёбра в вершины, которые не лежат макс. парсоче, я ставлю точку в эти вершины. Если их не осталось (значит все оставшиеся рёбра точно соединены с вершинами из макс. парсоча), ставлю все точки в оставшуюся левую (или правую) долю. Поддерживаю граф сетом и удаляю все рёбра из/в вершину, куда ставлю точку.
