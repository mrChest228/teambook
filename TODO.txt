bool isInRange(pair <ld, ld> p, ld a) {
    return p.f - eps <= p.s ? (p.f - eps <= a && a <= p.s + eps) : (p.f - eps <= a || a <= p.s + eps);
}

pair <ld, ld> mergeRanges(pair <ld, ld> a, pair <ld, ld> b) { // (a.s - a.f) % 2PI <= Pi, (b.s - b.f + 2PI) % 2PI <= PI
    if (isInRange(b, a.f))
        return min(pair <ld, ld>{a.f, a.s}, pair <ld, ld>{a.f, b.s}, [](pair <ld, ld> a, pair <ld, ld> b) {
            return a.s - a.f + 2 * PI * (a.s < a.f) < b.s - b.f + 2 * PI * (b.s < b.f);
        });
    if (isInRange(b, a.s))
        return min(pair <ld, ld>{a.f, a.s}, pair <ld, ld>{b.f, a.s}, [](pair <ld, ld> a, pair <ld, ld> b) {
            return a.s - a.f + 2 * PI * (a.s < a.f) < b.s - b.f + 2 * PI * (b.s < b.f);
        });
    if (isInRange(a, b.f))
        return b;
    return {-1, -1};
}

ld distToSegment(int x0, int y0, int x1, int y1, int x2, int y2) { // From (x0, y0) to (x1, y1)-(x2, y2)
    ld a = sqrtl(dist2(x0, y0, x1, y1)),
       b = sqrtl(dist2(x0, y0, x2, y2)),
       c = sqrtl(dist2(x1, y1, x2, y2));
    ld x = (b * b - a * a - c * c) / c / -2;
    if (0 <= x && x <= c)
        return sqrt(a * a - x * x);
    return 9e18; // Dist to point
}

const int B = 512;
struct Block { vector<u64> c; u64 lz = 0; gp_hash_table<u64, int, H> last; };
vector<Block> bl; int n = 0;

void rebuild(int k) {
    auto& b = bl[k]; b.last.clear();
    for (int j = 0; j < (int)b.c.size(); j++) b.last[b.c[j]] = k * B + j; // по возрастанию => остаётся максимум
}
void push(u64 h) {
    if (n % B == 0) bl.emplace_back();
    auto& b = bl.back(); u64 v = h ^ b.lz;
    b.c.push_back(v); b.last[v] = n++;
}
void suffixXor(int l, u64 x) {               // 0-индексация
    int k = l / B;
    for (int j = l - k * B; j < (int)bl[k].c.size(); j++) bl[k].c[j] ^= x;
    rebuild(k);
    for (int t = k + 1; t < (int)bl.size(); t++) bl[t].lz ^= x;
}
int findLast(u64 x) {
    for (int k = (int)bl.size() - 1; k >= 0; k--) {
        auto it = bl[k].last.find(x ^ bl[k].lz);
        if (it != bl[k].last.end()) return it->second;
    }
    return -1;
}
